# 数据库设计文档

## 1. 数据库概述

### 1.1 设计原则

#### 企业级数据库设计原则
1. **规范化设计**：遵循第三范式（3NF），减少数据冗余
2. **性能优化**：合理设计索引，优化查询性能
3. **数据完整性**：通过约束确保数据准确性
4. **安全性**：敏感数据加密，访问控制
5. **可扩展性**：预留扩展字段，支持业务增长

#### 数据库选型
- **主数据库**：MySQL 8.0 - 关系型数据存储
- **缓存数据库**：Redis 7.x - 高速缓存和会话存储
- **向量数据库**：Milvus 2.3 - 向量相似度搜索

### 3.3 缓存操作示例

```python
import redis
import json
from datetime import datetime, timedelta

class CacheManager:
    def __init__(self):
        self.redis = redis.Redis(
            host='localhost',
            port=6379,
            db=0,
            decode_responses=True
        )
    
    def set_user_session(self, session_id, user_data, ttl=3600):
        """设置用户会话"""
        key = f"session:{session_id}"
        self.redis.hset(key, mapping=user_data)
        self.redis.expire(key, ttl)
    
    def get_user_session(self, session_id):
        """获取用户会话"""
        key = f"session:{session_id}"
        return self.redis.hgetall(key)
    
    def cache_document(self, doc_id, doc_data, ttl=600):
        """缓存文档"""
        key = f"document:{doc_id}"
        self.redis.setex(key, ttl, json.dumps(doc_data))
    
    def get_cached_document(self, doc_id):
        """获取缓存文档"""
        key = f"document:{doc_id}"
        data = self.redis.get(key)
        return json.loads(data) if data else None
    
    def cache_search_results(self, query, results, ttl=300):
        """缓存搜索结果"""
        query_hash = hashlib.md5(query.encode()).hexdigest()
        key = f"search:{query_hash}"
        self.redis.setex(key, ttl, json.dumps(results))
    
    def add_to_hot_documents(self, doc_id, score=1):
        """添加到热门文档"""
        self.redis.zincrby("hot_documents", score, doc_id)
        # 保留前100个
        self.redis.zremrangebyrank("hot_documents", 0, -101)
    
    def get_hot_documents(self, limit=10):
        """获取热门文档"""
        return self.redis.zrevrange("hot_documents", 0, limit-1, withscores=True)
```

### 4.2 向量存储操作

```python
from pymilvus import connections, Collection
from langchain_openai import OpenAIEmbeddings

class VectorStore:
    def __init__(self):
        connections.connect("default", host="localhost", port="19530")
        self.collection = Collection("wiki_embeddings")
        self.embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
    
    def add_document(self, doc_id, chunks):
        """添加文档向量"""
        embeddings = self.embeddings.embed_documents([chunk["content"] for chunk in chunks])
        
        data = [
            [doc_id] * len(chunks),  # document_id
            list(range(len(chunks))),  # chunk_index
            [chunk["content"] for chunk in chunks],  # content
            embeddings,  # embedding
            [chunk.get("metadata", {}) for chunk in chunks]  # metadata
        ]
        
        self.collection.insert(data)
    
    def search(self, query, top_k=5, score_threshold=0.7):
        """向量搜索"""
        query_embedding = self.embeddings.embed_query(query)
        
        search_params = {
            "metric_type": "L2",
            "params": {"nprobe": 16}
        }
        
        results = self.collection.search(
            data=[query_embedding],
            anns_field="embedding",
            param=search_params,
            limit=top_k,
            output_fields=["document_id", "chunk_index", "content", "metadata"]
        )
        
        # 过滤低分结果
        filtered_results = []
        for hits in results:
            for hit in hits:
                if hit.score >= score_threshold:
                    filtered_results.append({
                        "document_id": hit.entity.get("document_id"),
                        "chunk_index": hit.entity.get("chunk_index"),
                        "content": hit.entity.get("content"),
                        "metadata": hit.entity.get("metadata"),
                        "score": hit.score
                    })
        
        return filtered_results
    
    def delete_document(self, doc_id):
        """删除文档向量"""
        self.collection.delete(f"document_id == {doc_id}")
```

```python
# 查询优化示例
class DocumentQueryOptimized:
    def search_with_pagination(self, keyword, category_id=None, page=1, per_page=20):
        """分页查询优化"""
        query = Document.query
        
        # 使用索引字段
        query = query.filter(Document.status == 'published')
        query = query.filter(Document.visibility == 'public')
        
        if keyword:
            # 使用全文索引
            query = query.filter(
                or_(
                    Document.title.ilike(f'%{keyword}%'),
                    Document.content.ilike(f'%{keyword}%')
                )
            )
        
        if category_id:
            query = query.filter(Document.category_id == category_id)
        
        # 只查询需要的字段
        query = query.with_entities(
            Document.id,
            Document.title,
            Document.summary,
            Document.view_count,
            Document.created_at
        )
        
        # 分页
        pagination = query.paginate(
            page=page,
            per_page=per_page,
            error_out=False
        )
        
        return {
            'items': pagination.items,
            'total': pagination.total,
            'pages': pagination.pages,
            'current_page': page
        }
```
