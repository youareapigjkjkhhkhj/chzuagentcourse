class ModelManager:
    def load_models(self):
        """从 train_model/ 目录加载所有可用模型"""
        model_dir = os.path.join(os.path.dirname(__file__), '..', 'train_model')
        for model_name in self.SUPPORTED_MODELS:
            weight_path = os.path.join(model_dir, model_name, 'best_model.pth')
            if os.path.exists(weight_path):
                model = timm.create_model(model_name, num_classes=5, pretrained=False)
                model.load_state_dict(torch.load(weight_path, map_location=self.device))
                self.models[model_name] = model

### 集成预测

```python
def ensemble_predict(self, image_tensor):
    """多模型加权集成预测"""
    # 1. 并行推理（ThreadPoolExecutor）
    predictions = {}
    with ThreadPoolExecutor(max_workers=4) as executor:
        futures = {
            executor.submit(self.predict_single_model, name, image_tensor): name
            for name in self.models
        }
        for future in futures:
            name = futures[future]
            predictions[name] = future.result()

    # 2. 加权概率平均
    ensemble_probs = np.zeros(5)
    total_weight = 0
    for name, pred in predictions.items():
        weight = self.MODEL_WEIGHTS[name]
        ensemble_probs += pred['probabilities'] * weight
        total_weight += weight
    ensemble_probs /= total_weight

    # 3. 最终分类
    final_level = int(np.argmax(ensemble_probs))
    confidence = float(ensemble_probs[final_level])

    return {
        'level': final_level,
        'confidence': confidence,
        'probabilities': ensemble_probs.tolist(),
        'model_results': predictions
    }
```
