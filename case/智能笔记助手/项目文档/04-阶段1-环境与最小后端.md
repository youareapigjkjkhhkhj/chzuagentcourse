## 1.4 核心代码

### 1.4.1 集中式配置 `src/config.py`

这是整个项目的「配置中枢」。关键点：`BASE_DIR` 由 `config.py` 自身位置反推（`backend/`），所有数据/向量目录都基于它绝对化，**不依赖当前工作目录**；实例化 `Settings` 时自动创建 `notes/`、`vector_store/` 目录。

```python
# 源码/backend/src/config.py:1-48（节选关键行）
import os
from pathlib import Path
from dotenv import load_dotenv

load_dotenv()
# config.py 位于 backend/src/，上溯两级得到 backend/ 作为项目根
BASE_DIR = Path(__file__).resolve().parent.parent          # 行 10
DEFAULT_DATA = BASE_DIR / "data"

class Settings:
    def __init__(self):
        self.BASE_DIR = BASE_DIR
        self.OPENAI_API_KEY = os.getenv("OPENAI_API_KEY", "")          # 行 22
        self.OPENAI_BASE_URL = os.getenv("OPENAI_BASE_URL", "https://api.openai.com/v1")
        self.CHAT_MODEL = os.getenv("CHAT_MODEL", "gpt-5-mini")        # 行 24
        self.EMBEDDING_MODEL = os.getenv("EMBEDDING_MODEL", "text-embedding-3-small")  # 行 25
        self.APP_HOST = os.getenv("APP_HOST", "0.0.0.0")
        self.APP_PORT = int(os.getenv("APP_PORT", "5000"))            # 行 29
        self.LOG_LEVEL = os.getenv("LOG_LEVEL", "INFO")
        # 路径（绝对化，避免 cwd 依赖）
        self.DATA_DIR = Path(os.getenv("DATA_DIR", str(DEFAULT_DATA)))           # 行 33
        self.NOTES_DIR = Path(os.getenv("NOTES_DIR", str(DEFAULT_DATA / "notes")))
        self.VECTOR_DIR = Path(os.getenv("VECTOR_DIR", str(DEFAULT_DATA / "vector_store")))
        self.FAISS_INDEX_NAME = os.getenv("FAISS_INDEX_NAME", "faiss_index")     # 行 36
        # RAG 参数
        self.CHUNK_SIZE = int(os.getenv("CHUNK_SIZE", "800"))            # 行 39
        self.CHUNK_OVERLAP = int(os.getenv("CHUNK_OVERLAP", "120"))
        self.TOP_K = int(os.getenv("TOP_K", "4"))                       # 行 41
        # 自动建目录
        self.NOTES_DIR.mkdir(parents=True, exist_ok=True)              # 行 44
        self.VECTOR_DIR.mkdir(parents=True, exist_ok=True)

settings = Settings()                                                # 行 48
```
