## 2.3 核心代码

### 2.3.1 文档加载器 `src/rag/loader.py`

统一把不同后缀的文件读成 LangChain `Document`，并写入 `metadata["source"]` / `["note_id"]` 供后续检索溯源。

```python
# 源码/backend/src/rag/loader.py:1-33
from pathlib import Path
from langchain_community.document_loaders import TextLoader, PyPDFLoader
from src.config import settings

SUPPORTED_EXT = {".txt", ".md", ".markdown", ".pdf"}     # 行 6

def is_supported(filename: str) -> bool:                 # 行 9
    return Path(filename).suffix.lower() in SUPPORTED_EXT

def load_document(path: str):                            # 行 14
    ext = Path(path).suffix.lower()
    if ext == ".pdf":
        loader = PyPDFLoader(path)
    elif ext in (".txt", ".md", ".markdown"):
        loader = TextLoader(path, encoding="utf-8", autodetect_encoding=True)  # 行 23
    else:
        raise ValueError(f"不支持的文件类型: {ext}")
    docs = loader.load()
    name = Path(path).name
    for d in docs:
        d.metadata["source"] = name                      # 行 30 溯源关键
        d.metadata["note_id"] = name
    return docs
```

### 2.3.2 笔记服务 `src/services/note_service.py`（本阶段相关部分）

文件管理：列表、安全保存、读取、删除。注意 `_safe_name` 只替换 `\/:*?"<>|` 等危险字符、**保留中文**。

```python
# 源码/backend/src/services/note_service.py:13-69（节选）
import re, time
from pathlib import Path
from src.config import settings
from src.rag.loader import load_document, is_supported

def _safe_name(filename: str) -> str:                    # 行 13
    name = Path(filename).name
    name = re.sub(r'[\\/:*?"<>|]', "_", name)            # 仅净化危险字符，保留中文
    if not name.strip():
        name = f"note_{int(time.time())}"
    return name

def list_notes() -> list:                                # 行 22
    notes = []
    if settings.NOTES_DIR.exists():
        for p in sorted(settings.NOTES_DIR.iterdir(), key=lambda x: x.stat().st_mtime, reverse=True):
            if p.is_file() and is_supported(p.name):
                notes.append({"id": p.name, "name": p.name,
                              "size": p.stat().st_size, "updated_at": int(p.stat().st_mtime)})
    return notes

def save_note(file_storage, original_filename: str) -> dict:   # 行 37
    name = _safe_name(original_filename)
    dest = settings.NOTES_DIR / name
    if dest.exists():                                    # 同名加时间戳避免覆盖
        stem, ext = os.path.splitext(name)
        name = f"{stem}_{int(time.time())}{ext}"
        dest = settings.NOTES_DIR / name
    file_storage.save(str(dest))
    return {"id": name, "name": name, "size": dest.stat().st_size, "updated_at": int(dest.stat().st_mtime)}

def get_note_content(note_id: str) -> str:               # 行 54
    p = settings.NOTES_DIR / note_id
    if not p.exists() or not p.is_file():
        raise FileNotFoundError(f"笔记不存在: {note_id}")
    return p.read_text(encoding="utf-8", errors="ignore")

def delete_note(note_id: str) -> bool:                  # 行 62
    p = settings.NOTES_DIR / note_id
    if p.exists() and p.is_file():
        p.unlink()
        VectorStoreManager().delete()                    # 删笔记后索引失效，需重建
        return True
    return False
```

### 2.3.3 笔记路由 `src/api/routes.py`（本阶段相关部分）

```python
# 源码/backend/src/api/routes.py:14-46（节选）
@api_bp.route("/notes", methods=["GET"])
def notes_list():
    return jsonify({"notes": note_service.list_notes()})

@api_bp.route("/notes/upload", methods=["POST"])
def notes_upload():
    if "file" not in request.files:
        return jsonify({"error": "缺少文件字段 file"}), 400
    f = request.files["file"]
    if not f.filename:
        return jsonify({"error": "文件名为空"}), 400
    try:
        meta = note_service.save_note(f, f.filename)
    except Exception as e:
        return jsonify({"error": str(e)}), 500
    return jsonify({"note": meta}), 201

@api_bp.route("/notes/<path:note_id>", methods=["GET"])
def notes_get(note_id):
    try:
        content = note_service.get_note_content(note_id)
    except FileNotFoundError as e:
        return jsonify({"error": str(e)}), 404
    return jsonify({"id": note_id, "content": content})

@api_bp.route("/notes/<path:note_id>", methods=["DELETE"])
def notes_delete(note_id):
    if not note_service.delete_note(note_id):
        return jsonify({"error": "笔记不存在"}), 404
    return jsonify({"deleted": note_id})
```
