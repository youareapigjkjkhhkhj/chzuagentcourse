### 6.3.1 流式问答 `src/services/qa_service.py`（新增 ask_stream）

```python
# 源码/backend/src/services/qa_service.py:9-11, 46-59（节选）
def _sse(payload: dict) -> str:                       # 行 9
    return f"data: {json.dumps(payload, ensure_ascii=False)}\n\n"

# ...（_retrieve / _sources 见阶段 5）

def ask_stream(self, question, top_k=None):           # 行 46
    """流式问答：逐 token 推送 SSE 事件，结束推送 sources。"""
    if not self.mgr.exists():
        yield _sse({"type": "error", "message": "知识库尚未构建，请先上传笔记并构建索引。"})
        return
    top_k = top_k or settings.TOP_K
    results = self._retrieve(question, top_k)
    context = format_docs([d for d, _ in results])
    llm = get_llm(streaming=True)                     # 行 54 开启流式
    messages = build_prompt().format_prompt(context=context, question=question).to_messages()
    for chunk in llm.stream(messages):                # 行 56 逐块产出
        if chunk.content:
            yield _sse({"type": "token", "content": chunk.content})
    yield _sse({"type": "done", "sources": self._sources(results)})  # 行 59 收尾来源
```

### 6.3.2 完整路由 `src/api/routes.py`

```python
# 源码/backend/src/api/routes.py:1-92（全，节选关键）
@api_bp.route("/index/rebuild", methods=["POST"])    # 行 50
def index_rebuild():
    try:
        result = note_service.build_index()
    except RuntimeError as e:
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        return jsonify({"error": str(e)}), 500
    return jsonify({"status": "built", **result})

@api_bp.route("/index/status", methods=["GET"])      # 行 61
def index_status():
    return jsonify(note_service.index_status())

@api_bp.route("/qa", methods=["POST"])               # 行 67 非流式
def qa():
    data = request.get_json(force=True, silent=True) or {}
    question = (data.get("question") or "").strip()
    if not question:
        return jsonify({"error": "问题不能为空"}), 400
    try:
        result = qa_service.QAService().ask(question, top_k=data.get("top_k"))  # 行 74
    except RuntimeError as e:
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        return jsonify({"error": str(e)}), 500
    return jsonify(result)

@api_bp.route("/qa/stream", methods=["POST"])        # 行 82 流式 SSE
def qa_stream():
    data = request.get_json(force=True, silent=True) or {}
    question = (data.get("question") or "").strip()
    if not question:
        return jsonify({"error": "问题不能为空"}), 400
    return Response(
        qa_service.QAService().ask_stream(question, top_k=data.get("top_k")),
        mimetype="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"})
```

```python
# 源码/backend/app.py:16-45（节选）
def create_app() -> Flask:
    app = Flask(__name__)
    CORS(app)                                        # 行 18 允许跨域（开发态 Vite 5173）
    app.register_blueprint(api_bp)                   # 行 19
    @app.route("/api/health")                        # 行 21
    @app.route("/health")
    def _health():
        return {"service": "智能笔记助手 RAG 后端", "version": "1.0.0"}
    # 可选：若前端已构建到 ../frontend/dist，则由 Flask 一并托管
    frontend_dist = settings.BASE_DIR.parent / "frontend" / "dist"   # 行 27
    if frontend_dist.exists():
        @app.route("/", defaults={"path": ""})       # 行 29
        @app.route("/<path:path>")
        def serve_frontend(path):
            target = frontend_dist / path
            if path and target.exists() and target.is_file():
                return send_from_directory(str(frontend_dist), path)
            return send_from_directory(str(frontend_dist), "index.html")

if __name__ == "__main__":
    app.run(host=settings.APP_HOST, port=settings.APP_PORT, debug=False, threaded=True)  # 行 45
```
