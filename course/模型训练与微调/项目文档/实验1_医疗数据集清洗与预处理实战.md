```
data/

    medical_sft_1K_format.jsonl  # 医疗SFT数据（ShareGPT格式）
    sharegpt_zh_1K_format.jsonl  # 通用中文对话数据

    dpo_zh_500.jsonl             # DPO偏好对数据
```

```json
{"conversations":[{"from":"human","value":"治疗阳痿吃什么药呢？"},{"from":"gpt","value":"男子早泄、早泄病症的再次发生..."}]}
```

### 步骤3：数据统计与质量分析
```python
# data_analysis.py - 医疗数据集质量分析脚本
import json
from collections import Counter

def analyze_dataset(filepath):
    """分析医疗数据集的基本统计信息"""
    total = 0
    empty_answers = 0
    duplicate_pairs = set()
    too_short = 0
    too_long = 0
    
    with open(filepath, 'r', encoding='utf-8') as f:
        for line in f:
            total += 1
            data = json.loads(line.strip())
            conversations = data.get('conversations', [])
            
            if len(conversations) < 2:
                empty_answers += 1
                continue
            
            human_text = conversations[0].get('value', '')
            gpt_text = conversations[1].get('value', '')
            
            pair_key = (human_text, gpt_text)
            if pair_key in duplicate_pairs:
                continue
            duplicate_pairs.add(pair_key)
            
            if len(gpt_text) < 10:
                too_short += 1
            if len(gpt_text) > 2000:
                too_long += 1
    
    print(f"数据集: {filepath}")
    print(f"  总样本数: {total}")
    print(f"  唯一样本数: {len(duplicate_pairs)}")
    print(f"  重复样本数: {total - len(duplicate_pairs)}")
    print(f"  空回答样本: {empty_answers}")
    print(f"  过短回答(<10字): {too_short}")
    print(f"  过长回答(>2000字): {too_long}")
    print(f"  质量合格率: {(len(duplicate_pairs) - too_short - too_long) / len(duplicate_pairs) * 100:.1f}%")

if __name__ == "__main__":
    analyze_dataset("data/sft/medical_sft_1K_format.jsonl")
    analyze_dataset("data/reward/dpo_zh_500.jsonl")
```

### 步骤4：数据清洗与过滤
```python
# data_clean.py - 医疗数据集清洗脚本
import json
import re

def is_valid_medical_text(text):
    """检查文本是否为有效的医疗文本"""
    if re.search(r'[\x00-\x08\x0b\x0c\x0e-\x1f]', text):
        return False
    if len(text.strip()) < 5:
        return False
    return True

def clean_medical_dataset(input_path, output_path):
    """清洗医疗数据集"""
    seen = set()
    cleaned_count = 0
    
    with open(input_path, 'r', encoding='utf-8') as fin, \
         open(output_path, 'w', encoding='utf-8') as fout:
        for line in fin:
            data = json.loads(line.strip())
            conversations = data.get('conversations', [])
            
            if len(conversations) < 2:
                continue
            
            human_text = conversations[0].get('value', '')
            gpt_text = conversations[1].get('value', '')
            
            pair_key = (human_text, gpt_text)
            if pair_key in seen:
                continue
            seen.add(pair_key)
            
            if not is_valid_medical_text(human_text) or not is_valid_medical_text(gpt_text):
                continue
            
            cleaned_data = {"conversations": conversations}
            fout.write(json.dumps(cleaned_data, ensure_ascii=False) + '\n')
            cleaned_count += 1
    
    print(f"清洗完成: {input_path} -> {output_path}")
    print(f"  清洗后样本数: {cleaned_count}")

if __name__ == "__main__":
    clean_medical_dataset(
        "data/sft/medical_sft_1K_format.jsonl",
        "data/sft/medical_sft_cleaned.jsonl"
    )
```

### 步骤5：划分训练集/验证集
```python
# data_split.py - 划分训练集和验证集
import json
import random

def split_dataset(input_path, train_path, val_path, val_ratio=0.1, seed=42):
    random.seed(seed)
    
    with open(input_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()
    
    random.shuffle(lines)
    split_idx = int(len(lines) * (1 - val_ratio))
    
    with open(train_path, 'w', encoding='utf-8') as f:
        f.writelines(lines[:split_idx])
    
    with open(val_path, 'w', encoding='utf-8') as f:
        f.writelines(lines[split_idx:])
    
    print(f"训练集: {train_path} ({split_idx} 样本)")
    print(f"验证集: {val_path} ({len(lines) - split_idx} 样本)")

if __name__ == "__main__":
    split_dataset(
        "data/sft/medical_sft_cleaned.jsonl",
        "data/sft/train/medical_sft_train.jsonl",
        "data/sft/val/medical_sft_val.jsonl"
    )
```
