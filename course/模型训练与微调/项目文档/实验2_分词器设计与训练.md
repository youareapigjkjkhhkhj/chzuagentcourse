> **类型**：⚠️拓展实验，需补充代码 | **难度**：★★★☆☆ | **时长**：2小时

## 实验目的
理解分词器原理，针对医疗专业术语，学习扩展/训练分词器。

## 实验原理
Qwen2‑4B‑Instruct使用BPE（Byte-Pair Encoding）分词器。医疗领域包含大量专业术语（如"心肌梗死"、"冠状动脉"），通用分词器可能将这些术语切分成多个子词，影响模型对医疗概念的理解。

> **说明**：MedicalGPT仓库无此脚本，需要独立基于`tokenizers`库编码，适合拓展。

## 实验内容

### 步骤1：分析现有分词器对医疗术语的切分效果
```python
# tokenizer_analysis.py - 分析Qwen分词器对医疗术语的切分
from transformers import AutoTokenizer

def analyze_tokenizer(model_path="Qwen/Qwen2.5-0.5B-Instruct"):
    """分析分词器对医疗术语的切分效果"""
    tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
    
    medical_terms = [
        "心肌梗死", "冠状动脉粥样硬化", "高血压",
        "糖尿病", "白血病", "肺炎", "肝硬化",
        "脑卒中", "心律失常", "肺栓塞",
        "阿司匹林", "布洛芬", "青霉素"
    ]
    
    print(f"分词器词汇表大小: {len(tokenizer)}")
    print(f"\n医疗术语切分分析:")
    print("-" * 60)
    
    for term in medical_terms:
        tokens = tokenizer.tokenize(term)
        token_ids = tokenizer.encode(term)
        print(f"术语: {term}")
        print(f"  切分结果: {tokens}")
        print(f"  Token数量: {len(tokens)}")
        print()

if __name__ == "__main__":
    analyze_tokenizer()
```

### 步骤2：收集医疗专业词汇并添加到分词器
```python
# tokenizer_extend.py - 扩展分词器添加医疗词汇
from transformers import AutoTokenizer

def extend_tokenizer_with_medical_terms(model_path, new_vocab_path, output_path):
    """向分词器添加医疗专业词汇"""
    tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
    
    with open(new_vocab_path, 'r', encoding='utf-8') as f:
        medical_vocab = [line.strip() for line in f if line.strip()]
    
    print(f"原始词汇表大小: {len(tokenizer)}")
    print(f"待添加医疗词汇数: {len(medical_vocab)}")
    
    num_added = tokenizer.add_tokens(medical_vocab)
    print(f"成功添加: {num_added} 个新词")
    
    tokenizer.save_pretrained(output_path)
    print(f"扩展后的分词器已保存到: {output_path}")
    
    return tokenizer

if __name__ == "__main__":
    extend_tokenizer_with_medical_terms(
        model_path="Qwen/Qwen2.5-0.5B-Instruct",
        new_vocab_path="data/vocab/medical_vocab.txt",
        output_path="tokenizer-medical"
    )
```

### 步骤3：医疗词汇表构建
```python
# build_medical_vocab.py - 从医疗语料中提取专业词汇
import jieba
from collections import Counter

def build_medical_vocab(corpus_path, output_path, min_freq=5):
    """从医疗语料中提取高频专业词汇"""
    word_counter = Counter()
    
    with open(corpus_path, 'r', encoding='utf-8') as f:
        for line in f:
            words = jieba.cut(line.strip())
            word_counter.update(words)
    
    medical_words = [
        word for word, freq in word_counter.most_common()
        if freq >= min_freq and len(word) >= 2
    ]
    
    with open(output_path, 'w', encoding='utf-8') as f:
        for word in medical_words:
            f.write(word + '\n')
    
    print(f"构建医疗词汇表完成: {output_path}")
    print(f"词汇数量: {len(medical_words)}")

if __name__ == "__main__":
    build_medical_vocab(
        corpus_path="data/pretrain/fever.txt",
        output_path="data/vocab/medical_vocab.txt",
        min_freq=3
    )
```
