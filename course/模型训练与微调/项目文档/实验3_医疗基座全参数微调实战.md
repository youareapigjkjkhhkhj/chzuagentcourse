# 实验3 医疗基座全参数微调实战

> **类型**：✅必做实验 | **难度**：★★★☆☆ | **时长**：3小时

## 实验目的
理解全参数微调原理，对比全参微调显存开销与训练效果，完成Qwen2‑4B医疗场景全参数SFT。

## 实验原理
全参数微调（Full Parameter Fine-tuning）更新模型所有参数。MedicalGPT的SFT脚本通过`--use_peft False`关闭LoRA，启用全参数训练。

## 实验脚本与参数
- **脚本路径**：`training/supervised_finetuning.py`
- **关键参数**：`--use_peft False`

## 实验内容

### 步骤1：修改启动脚本，关闭LoRA
```bash
# run_sft_full.sh - 全参数微调启动脚本
CUDA_VISIBLE_DEVICES=0,1 torchrun --nproc_per_node 2 training/supervised_finetuning.py \
    --model_name_or_path Qwen/Qwen2.5-4B-Instruct \
    --train_file_dir ./data/sft \
    --validation_file_dir ./data/sft \
    --per_device_train_batch_size 1 \
    --per_device_eval_batch_size 1 \
    --do_train \
    --do_eval \
    --use_peft False \
    --max_train_samples 500 \
    --max_eval_samples 10 \
    --model_max_length 512 \
    --num_train_epochs 1 \
    --learning_rate 2e-5 \
    --warmup_steps 5 \
    --weight_decay 0.05 \
    --logging_strategy steps \
    --logging_steps 10 \
    --eval_steps 50 \
    --eval_strategy steps \
    --save_steps 500 \
    --save_strategy steps \
    --save_total_limit 3 \
    --gradient_accumulation_steps 16 \
    --preprocessing_num_workers 4 \
    --output_dir outputs-sft-full-qwen \
    --ddp_timeout 30000 \
    --logging_first_step True \
    --torch_dtype bfloat16 \
    --bf16 \
    --report_to tensorboard \
    --ddp_find_unused_parameters False \
    --gradient_checkpointing True \
    --cache_dir ./cache \
    --flash_attn True
```

> **关键参数说明**：
> - `--use_peft False`：关闭LoRA，启用全参数微调
> - `--per_device_train_batch_size 1`：全参微调显存占用大，batch_size需减小
> - `--gradient_accumulation_steps 16`：通过梯度累积等效增大batch_size
> - `--gradient_checkpointing True`：启用梯度检查点，节省显存

### 步骤3：记录训练指标
```python
# monitor_training.py - 监控训练过程
from tensorboard.backend.event_processing.event_accumulator import EventAccumulator
import matplotlib.pyplot as plt

def plot_training_loss(log_dir):
    ea = EventAccumulator(log_dir)
    ea.Reload()
    
    loss_events = ea.Scalars('train/loss')
    steps = [e.step for e in loss_events]
    values = [e.value for e in loss_events]
    
    plt.figure(figsize=(10, 6))
    plt.plot(steps, values)
    plt.xlabel('Steps')
    plt.ylabel('Loss')
    plt.title('Full Parameter Fine-tuning Training Loss')
    plt.savefig('loss_full_ft.png')
    print(f"Loss曲线已保存: loss_full_ft.png")

if __name__ == "__main__":
    plot_training_loss("outputs-sft-full-qwen/runs/")
```

### 步骤4：医疗问答效果测试
```python
# test_medical_qa.py - 测试医疗问答效果
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

def test_medical_qa(model_path, test_queries):
    tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
    model = AutoModelForCausalLM.from_pretrained(
        model_path, torch_dtype=torch.bfloat16, device_map="auto", trust_remote_code=True
    )
    
    for query in test_queries:
        messages = [
            {"role": "system", "content": "你是一个专业的医疗科普助手。"},
            {"role": "user", "content": query}
        ]
        text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
        inputs = tokenizer(text, return_tensors="pt").to(model.device)
        
        with torch.no_grad():
            outputs = model.generate(**inputs, max_new_tokens=512, temperature=0.7)
        
        response = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
        print(f"问题: {query}")
        print(f"回答: {response}")
        print("-" * 80)

if __name__ == "__main__":
    test_queries = [
        "感冒需要吃什么药？",
        "高血压患者日常饮食需要注意什么？",
        "糖尿病的早期症状有哪些？"
    ]
    test_medical_qa("outputs-sft-full-qwen", test_queries)
```
